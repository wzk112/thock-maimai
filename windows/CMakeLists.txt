cmake_minimum_required(VERSION 3.24)
project(ThockMaimai VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  add_executable(core-test core_test.cpp)
  enable_testing()
  add_test(NAME core COMMAND core-test)
  return()
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(ThockMaimai WIN32 main.cpp renderer.cpp app.manifest)
target_compile_definitions(ThockMaimai PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(ThockMaimai PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(ThockMaimai PRIVATE d3d11 dxgi d3dcompiler dcomp windowscodecs xaudio2 ole32 shell32 comctl32 wtsapi32)
set_target_properties(ThockMaimai PROPERTIES OUTPUT_NAME "Thock maimai")
add_executable(core-test core_test.cpp)
enable_testing()
add_test(NAME core COMMAND core-test)
add_custom_command(TARGET ThockMaimai POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  ${CMAKE_CURRENT_SOURCE_DIR}/effect.hlsl "$<TARGET_FILE_DIR:ThockMaimai>/effect.hlsl")
